# Run Application

    docker-compose up -d

# Verify

    docker-compose ps

    docker ps

    docker volume ls

    localhost:8080

# Stop Application

    docker-compose stop

# Start Again to Verify Logs

    docker-compose start

    docker-compose logs

    docker-compose logs -f wordpress

# Tearing down and Restoring

    docker-compose down

    docker-compose ps

    docker volume ls

    docker-compose up -d

    localhost:8080
